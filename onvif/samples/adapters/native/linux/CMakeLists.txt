cmake_minimum_required(VERSION 3.20)
project(linux_camera_capture LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux worker requires a real Linux toolchain and headers. Use tools/tests/adapters/linux for portable-only tests.")
endif()
option(CAPTURE_WITH_ARAVIS "Build pinned Aravis 0.8.36 GigE/Fake backends" ON)
option(CAPTURE_WITH_ARAVIS_USB "Build explicitly permitted stock Aravis USB path" ON)
option(CAPTURE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(PkgConfig REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
find_package(JPEG REQUIRED)
include(CheckIncludeFileCXX)
check_include_file_cxx(linux/videodev2.h HAVE_V4L2)
if(NOT HAVE_V4L2)
  message(FATAL_ERROR "Linux V4L2 UAPI headers are required; no stub capture backend is built.")
endif()
set(COMMON "${CMAKE_CURRENT_LIST_DIR}/../common")
set(TESTS "${CMAKE_CURRENT_LIST_DIR}/../../../../tools/tests/adapters/linux")
add_library(camera-linux-support STATIC "${COMMON}/bounds.cpp" runtime.cpp jpeg.cpp)
target_include_directories(camera-linux-support PUBLIC "${COMMON}")
target_compile_features(camera-linux-support PUBLIC cxx_std_17)
target_link_libraries(camera-linux-support PUBLIC OpenSSL::Crypto JPEG::JPEG)
add_executable(linux-capture-worker worker.cpp "../../uvc/linux/v4l2.cpp")
target_link_libraries(linux-capture-worker PRIVATE camera-linux-support)
target_compile_definitions(linux-capture-worker PRIVATE _FILE_OFFSET_BITS=64)
if(CAPTURE_WITH_ARAVIS)
  pkg_check_modules(ARAVIS REQUIRED IMPORTED_TARGET aravis-0.8=0.8.36)
  target_sources(linux-capture-worker PRIVATE "../../genicam/aravis.cpp")
  target_link_libraries(linux-capture-worker PRIVATE PkgConfig::ARAVIS)
  target_compile_definitions(linux-capture-worker PRIVATE CAPTURE_ARAVIS=1)
  if(CAPTURE_WITH_ARAVIS_USB)
    pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0>=1.0.16)
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_LIBRARIES PkgConfig::ARAVIS)
    check_cxx_source_compiles("#include <arv.h>\n#if !ARAVIS_HAS_USB\n#error USB support required\n#endif\nint main(){return 0;}" HAVE_ARAVIS_USB)
    unset(CMAKE_REQUIRED_LIBRARIES)
    if(NOT HAVE_ARAVIS_USB)
      message(FATAL_ERROR "Aravis was built without USB. Explicitly configure CAPTURE_WITH_ARAVIS_USB=OFF.")
    endif()
    target_link_libraries(linux-capture-worker PRIVATE PkgConfig::LIBUSB)
    target_compile_definitions(linux-capture-worker PRIVATE CAPTURE_ARAVIS_USB=1)
  endif()
else()
  message(STATUS "Aravis explicitly disabled: all Aravis selections return MissingDependency.")
endif()
foreach(target camera-linux-support linux-capture-worker)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CAPTURE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_subdirectory("${TESTS}" "${CMAKE_CURRENT_BINARY_DIR}/portable-tests")
add_executable(camera-jpeg-tests "${TESTS}/jpeg.cpp")
target_link_libraries(camera-jpeg-tests PRIVATE camera-linux-support)
add_test(NAME camera-system-jpeg COMMAND camera-jpeg-tests)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_test(NAME camera-protocol-no-device
  COMMAND "${Python3_EXECUTABLE}" "${TESTS}/protocol.py" --worker $<TARGET_FILE:linux-capture-worker>)
if(CAPTURE_WITH_ARAVIS)
  add_test(NAME camera-aravis-sdk-fake
    COMMAND "${Python3_EXECUTABLE}" "${TESTS}/protocol.py" --worker $<TARGET_FILE:linux-capture-worker> --fake)
  set_tests_properties(camera-aravis-sdk-fake PROPERTIES TIMEOUT 30)
endif()
