cmake_minimum_required(VERSION 3.20)
project(camera_portable_tests LANGUAGES CXX)
enable_testing()
set(COMMON "${CMAKE_CURRENT_LIST_DIR}/../../../../samples/adapters/native/common")
add_executable(camera-portable-tests portable.cpp "${COMMON}/bounds.cpp")
target_compile_features(camera-portable-tests PRIVATE cxx_std_17)
target_include_directories(camera-portable-tests PRIVATE "${COMMON}")
if(MSVC)
  target_compile_options(camera-portable-tests PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(camera-portable-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME camera-portable-bounds COMMAND camera-portable-tests)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND TARGET camera-linux-support)
  add_executable(camera-v4l2-fixture v4l2-fixture.cpp
    "${COMMON}/../../uvc/linux/v4l2.cpp")
  target_link_libraries(camera-v4l2-fixture PRIVATE camera-linux-support)
  target_compile_definitions(camera-v4l2-fixture PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
  foreach(symbol open64 stat64 fstat64 realpath ioctl poll close mmap64 munmap)
    target_link_options(camera-v4l2-fixture PRIVATE "-Wl,--wrap=${symbol}")
  endforeach()
  add_test(NAME camera-v4l2-simulated-ioctl COMMAND camera-v4l2-fixture)
  set_tests_properties(camera-v4l2-simulated-ioctl PROPERTIES TIMEOUT 10)
  if(CAPTURE_WITH_ARAVIS)
    add_executable(camera-aravis-contract aravis-contract.cpp
      "${COMMON}/../../genicam/aravis.cpp")
    target_link_libraries(camera-aravis-contract PRIVATE camera-linux-support PkgConfig::ARAVIS)
    target_compile_definitions(camera-aravis-contract PRIVATE
      CAPTURE_ARAVIS=1 CAPTURE_ARAVIS_USB=$<BOOL:${CAPTURE_WITH_ARAVIS_USB}>)
    if(CAPTURE_WITH_ARAVIS_USB)
      target_link_libraries(camera-aravis-contract PRIVATE PkgConfig::LIBUSB)
    endif()
    add_test(NAME camera-aravis-api-contract COMMAND camera-aravis-contract)
    set_tests_properties(camera-aravis-api-contract PROPERTIES TIMEOUT 15)
  endif()
  foreach(target camera-v4l2-fixture camera-aravis-contract)
    if(TARGET ${target})
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
      if(CAPTURE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
  endforeach()
endif()
